Passo 2 — Apps Script Extensões → Apps Script. Apague tudo e cole o código abaixo: 
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var d  = JSON.parse(e.postData.contents);

    if (d.tipo === 'encomenda') {
      var se = ss.getSheetByName('Encomendas');
      if (se.getLastRow() === 0)
        se.appendRow(['Data','Hora','Cliente','Telefone',
          'Descricao','Operador','Cidade','Dispositivo','Canal']);
      se.appendRow([d.data,d.hora,d.nome,d.telefone,d.descricao,
        d.operador,d.cidade,d.dispositivo,d.canal]);
    } else {
      var sv = ss.getSheetByName('Vendas');
      if (sv.getLastRow() === 0)
        sv.appendRow(['Data','Hora','Cliente','Telefone',
          'Produto','Qtd','Valor Unit.','Total',
          'Dinheiro','Pix','Credito','Debito','Troco',
          'Operador','Cidade','Dispositivo','Canal']);
      sv.appendRow([
        d.data,d.hora,d.nome,d.telefone,
        d.produto,d.qtd,d.valorUnit,d.total,
        d.pagamentos.Dinheiro||0,d.pagamentos.Pix||0,
        d.pagamentos.Credito||0,d.pagamentos.Debito||0,
        d.troco||0,
        d.operador,d.cidade,d.dispositivo,d.canal
      ]);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,erro:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
