Passo 1 — Crie a planilha Acesse sheets.google.com, crie uma nova planilha e nomeie a primeira aba como Vendas.
Passo 2 — Abra o Apps Script No menu da planilha: Extensões → Apps Script. Apague o código existente e cole o código abaixo: 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName('Vendas');

    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'Data','Hora','Cliente',
        'Telefone','Produto','Qtd',
        'Valor Unit.','Total','Pagamento'
      ]);
    }

    var d = JSON.parse(e.postData.contents);
    sheet.appendRow([
      d.data, d.hora, d.nome,
      d.telefone, d.produto, d.qtd,
      d.valorUnit, d.total, d.pagamento
    ]);

    return ContentService
      .createTextOutput(
        JSON.stringify({ok: true})
      )
      .setMimeType(
        ContentService.MimeType.JSON
      );
  } catch(err) {
    return ContentService
      .createTextOutput(
        JSON.stringify({ok:false,
          erro: err.toString()})
      )
      .setMimeType(
        ContentService.MimeType.JSON
      );
  }
}
