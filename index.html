function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var d  = JSON.parse(e.postData.contents);

    if (d.tipo === 'wa_update') {
      var sv2 = ss.getSheetByName('Vendas');
      var last = sv2.getLastRow();
      if (last > 1) {
        sv2.getRange(last, 15).setValue(d.waCliqueHora||'');
        sv2.getRange(last, 16).setValue(d.waConfirmadoHora||'');
      }

    } else if (d.tipo === 'encomenda') {
      var se = ss.getSheetByName('Encomendas');
      if (se.getLastRow() === 0)
        se.appendRow(['Data','Hora','Telefone','Cliente',
          'Descricao','Operador','Cidade','Dispositivo']);
      se.appendRow([d.data,d.hora,d.telefone,d.nome,
        d.descricao,d.operador,d.cidade,d.dispositivo]);

    } else {
      var sv = ss.getSheetByName('Vendas');
      if (sv.getLastRow() === 0)
        sv.appendRow(['Data','Hora','Telefone','Cliente',
          'Produtos','Total','Dinheiro','Pix',
          'Credito','Debito','Troco',
          'Operador','Cidade','Dispositivo',
          'WA_Enviado_Hora','WA_Confirmado_Hora']);
      sv.appendRow([
        d.data,d.hora,d.telefone,d.nome,
        d.produtosTexto,d.total,
        d.pagamentos.Dinheiro||0,
        d.pagamentos.Pix||0,
        d.pagamentos.Credito||0,
        d.pagamentos.Debito||0,
        d.troco||0,
        d.operador,d.cidade,d.dispositivo,
        '.',''
      ]);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,erro:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
